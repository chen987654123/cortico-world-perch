/**
 * Local patch (cortico-world-perch): the window tops the pet may stand on. They come from the
 * window process itself (`pet:windows`), which converts them against this pet window's own origin
 * and display, so a second monitor at a negative origin or with another scale needs no work here.
 */
let terrain = [];
let featuresKey = '';
let windowRects = [];
// A World command may arrive before the iframe body is ready (or while Coo is being dragged).
// Keep it until the body accepts it instead of consuming it as a silent no-op.
let pendingThunderJump = null;
/** Where the last accepted jump was aimed, for the World's status and for checking the chain. */
let lastPick = null;
/** Why the last order did or did not become a jump: the World shows this instead of guessing. */
let jumpDebug = { seen: 0, last: null, why: null };
// `take=1`: only the pet window runs the orders. The settings window's preview page runs this same
// script and would otherwise swallow a jump meant for the pet, leaving a silent no-op.
const TERRAIN_URL = 'http://127.0.0.1:47650/api/terrain' + (host ? '?take=1' : '');
/** Identifies this page load in the World's log, so a jump can be traced to the page that took it. */
const PAGE_ID = Math.random().toString(36).slice(2, 8);
const PAGE_HEADERS = { 'x-perch-page': PAGE_ID, 'x-perch-host': host ? '1' : '0' };
async function pollTerrain() {
  try {
    const res = await fetch(TERRAIN_URL, { cache: 'no-store', headers: PAGE_HEADERS });
    const data = await res.json();
    // The World sends jump targets in physical screen pixels; these two turn them back into page
    // coordinates. The tops themselves no longer come from here: the window process knows them
    // better, because it converts them against this window's own origin and display.
    const d = petScreen?.scaleFactor || devicePixelRatio || 1;
    const ox = data?.area?.x ?? petScreen?.workArea?.x ?? 0, oy = data?.area?.y ?? petScreen?.workArea?.y ?? 0;
    const featureSettings = data?.settings ?? {};
    // only remember the settings as delivered once a body was there to take them: the pack may
    // still be loading, and a set() swallowed then would never be retried
    const key = JSON.stringify(featureSettings);
    if (key !== featuresKey && body) {
      featuresKey = key;
      body.set({ bounds: bounds(), perchFeatures: featureSettings });
    }
    // Orders from the perch World. The World sends `command` as an object
    // ({ command: 'hop', target, apex }); older callers sent a bare string, so accept both —
    // comparing the object to 'hop' silently dropped every jump.
    const order = data.command && typeof data.command === 'object'
      ? data.command
      : data.command ? { command: data.command } : null;
    const kind = order?.command ?? null;
    if (kind) {
      jumpDebug.seen += 1;
      jumpDebug.last = kind;
      console.log(`[perch] order ${kind} target=${JSON.stringify(order.target)} body=${!!body} thunder=${typeof body?.thunderJump} mode=${body?.layout?.mode}`);
    }
    if (kind === 'hop') {
      if (featureSettings.aiThunderJump === false) jumpDebug.why = 'aiThunderJump off';
      else pendingThunderJump = { kind: 'hop', target: order.target, apex: Number(order.apex) || 180 };
    } else if (kind === 'jump') {
      pendingThunderJump = { kind: 'jump', x: Number(order.x), y: Number(order.y), apex: Number(order.apex) || 180 };
    }
    if (pendingThunderJump) {
      if (featureSettings.aiThunderJump === false) jumpDebug.why = 'aiThunderJump off';
      else if (!body) jumpDebug.why = 'body not loaded';
      else if (!body.thunderJump) jumpDebug.why = 'body has no thunderJump';
      else if (body.layout?.mode === 'drag') jumpDebug.why = 'being dragged';
      else jumpDebug.why = 'ready to jump';
    }
    if (featureSettings.aiThunderJump !== false && pendingThunderJump && body?.thunderJump && body.layout?.mode !== 'drag') {
      const cmd = pendingThunderJump;
      const here = body.layout;
      const floor = innerHeight - floorGap();
      const feet = here?.box ? here.box.y + here.box.h : floor;
      const x0 = here?.x ?? innerWidth / 2;
      let pick = null;
      if (cmd.kind === 'jump') {
        pick = { x: cmd.x, y: cmd.y };
      } else if (cmd.target && typeof cmd.target === 'object' && Number.isFinite(Number(cmd.target.x)) && Number.isFinite(Number(cmd.target.y))) {
        // an explicit destination, in physical screen pixels: convert it to this page's coordinates
        pick = { x: (Number(cmd.target.x) - ox) / d, y: (Number(cmd.target.y) - oy) / d };
      } else {
        // choose a top wide enough to land on, other than the one already underfoot, and land
        // away from where the pet stands: a spot straight above is a bob on the spot, not a leap
        const tops = [];
        for (const p of terrain) {
          const span = (p.spans ?? []).filter(([a, b]) => b - a >= 90).sort((a, b) => (b[1] - b[0]) - (a[1] - a[0]))[0];
          if (!span) continue;
          if (Math.abs(p.y - feet) <= 40) continue;
          const lo = span[0] + 28, hi = span[1] - 28;
          const far = Math.abs(lo - x0) >= Math.abs(hi - x0) ? lo : hi;
          const near = Math.abs(lo - x0) < Math.abs(hi - x0) ? lo : hi;
          const x = Math.abs(far - x0) >= 120 ? far : Math.abs(near - x0) >= 120 ? near : Math.max(lo, Math.min(hi, x0));
          tops.push({ x, y: p.y });
        }
        const targets = (cmd.target === 'floor' || featureSettings.autoWindowHops === false) ? [] : tops;
        const onFloor = feet >= floor - 6;
        const away = (min) => Math.max(30, Math.min(innerWidth - 30, x0 + (x0 < innerWidth / 2 ? 1 : -1) * min));
        pick = targets.sort((a, b) => Math.abs(a.y - feet) + Math.abs(a.x - x0) * .3 - (Math.abs(b.y - feet) + Math.abs(b.x - x0) * .3))[0]
          ?? (onFloor ? { x: away(280), y: floor } : { x: away(140), y: floor });
        if (pick.y === floor && Math.abs(pick.x - x0) < 150) pick.x = away(240);
      }
      // thunderJump returns false while the body is busy/dragging or the arc is invalid.
      // Only clear the queued command after a real jump was accepted.
      if (pick && body.thunderJump(pick.x, pick.y, cmd.apex) !== false) {
        lastPick = { fromX: Math.round(x0), fromY: Math.round(feet), toX: Math.round(pick.x), toY: Math.round(pick.y), apex: cmd.apex, at: Date.now() };
        pendingThunderJump = null;
        jumpDebug.why = 'sent';
      } else if (pick) {
        jumpDebug.why = 'body declined';
      } else {
        jumpDebug.why = 'no target';
      }
    }
  } catch { /* the perch World is off: the pet just walks the floor */ }
}
setInterval(pollTerrain, 500);
pollTerrain();
// The window tops, straight from the window process. Every ledge is already in this page's
// pixels — the window process subtracted this pet window's own origin and clipped to the display
// the window covers — so a second monitor needs no conversion and no guessing about which one
// the pet is on.
host?.onWindows?.((list) => {
  if (!Array.isArray(list)) return;
  const tops = [];
  for (const l of list) {
    const y = Number(l?.y), a = Number(l?.x0), b = Number(l?.x1);
    if (!Number.isFinite(y) || !Number.isFinite(a) || !Number.isFinite(b) || b <= a) continue;
    const last = tops[tops.length - 1];
    if (last && last.y === y) last.spans.push([a, b]);
    else tops.push({ y, spans: [[a, b]] });
  }
  terrain = tops;
  windowRects = [];
  body?.set({ bounds: bounds() });
});
// Coo's pack does not declare `needs: windows`, so the desktop pet keeps that sampler off. This
// World does want the tops, so it turns the same sampler on; the call is cheap and idempotent.
if (host) setInterval(() => host.watchWindows?.(true), 4000);
// where the body is, for the perch World's status tool (and for checking the chain from outside)
function reportState() {
  // only the pet window reports where the pet is; a preview tab has its own body on another stage
  if (!host) return;
  const l = body?.layout;
  // A body that never mounts would leave the World without a position, and then every window top is
  // measured against the wrong monitor. Report the reason once instead of going silent.
  const why = l ? '' : body ? 'no-layout' : 'no-body';
  if (!l && reportState.lastWhy === why) return;
  reportState.lastWhy = why;
  fetch('http://127.0.0.1:47650/api/pet', {
    method: 'POST', headers: { 'content-type': 'application/json' },
    body: JSON.stringify({
      x: Math.round(l?.x ?? NaN),
      y: Math.round(l?.box ? l.box.y + l.box.h : NaN),
      screen: petScreen ? { x: petScreen.x, y: petScreen.y, scaleFactor: petScreen.scaleFactor } : null,
      mode: l?.mode ?? null,
      terrain: terrain.length,
      jump: body?.lastJump ?? null,
      pick: lastPick,
      roam: l?.roam ?? null,
      diag: l?.diag ?? null,
      debug: jumpDebug,
      why,
      page: PAGE_ID,
    }),
  }).catch(() => {});
}
setInterval(reportState, 700);
